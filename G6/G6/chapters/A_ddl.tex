
\chapter{DDL tham khảo}
\label{app:ddl}

\section{Lưu ý}
DDL dưới đây được dựng từ các bảng/cột thể hiện trong các sơ đồ thiết kế. Đây là \textbf{DDL tham khảo}, chưa được tuyên bố là script triển khai chính thức. Trước khi chạy, nhóm cần chọn DBMS cụ thể và điều chỉnh cú pháp nếu cần.

\begin{lstlisting}[language=SQL]
CREATE TABLE ROLE (
  RoleID VARCHAR(20) PRIMARY KEY,
  RoleName VARCHAR(50) NOT NULL
);

CREATE TABLE USER (
  UserID VARCHAR(20) PRIMARY KEY,
  RoleID VARCHAR(20) NOT NULL,
  Username VARCHAR(50) NOT NULL,
  Email VARCHAR(100) NOT NULL,
  PasswordHash VARCHAR(255) NOT NULL,
  FullName VARCHAR(100),
  Status VARCHAR(20),
  CreatedAt DATETIME,
  LastLoginAt DATETIME,
  CONSTRAINT FK_USER_ROLE
    FOREIGN KEY (RoleID) REFERENCES ROLE(RoleID)
);

CREATE TABLE FARM (
  FarmID VARCHAR(20) PRIMARY KEY,
  FarmName VARCHAR(100) NOT NULL,
  Description TEXT,
  Address VARCHAR(255),
  Latitude DECIMAL(10,8),
  Longitude DECIMAL(11,8),
  Status VARCHAR(20),
  CreatedAt DATETIME
);

CREATE TABLE ZONE (
  ZoneID VARCHAR(20) PRIMARY KEY,
  FarmID VARCHAR(20) NOT NULL,
  ZoneName VARCHAR(100) NOT NULL,
  Description TEXT,
  Latitude DECIMAL(10,8),
  Longitude DECIMAL(11,8),
  Status VARCHAR(20),
  CONSTRAINT FK_ZONE_FARM
    FOREIGN KEY (FarmID) REFERENCES FARM(FarmID)
);

CREATE TABLE SENSOR_TYPE (
  SensorTypeID VARCHAR(20) PRIMARY KEY,
  TypeName VARCHAR(50) NOT NULL,
  Unit VARCHAR(20),
  Description TEXT
);

CREATE TABLE SENSOR_NODE (
  SensorID VARCHAR(20) PRIMARY KEY,
  ZoneID VARCHAR(20) NOT NULL,
  SensorCode VARCHAR(50),
  SerialNumber VARCHAR(50),
  Latitude DECIMAL(10,8),
  Longitude DECIMAL(11,8),
  BatteryLevel DECIMAL(5,2),
  Status VARCHAR(20),
  RegisteredAt DATETIME,
  LastSeenAt DATETIME,
  CONSTRAINT FK_SENSOR_ZONE
    FOREIGN KEY (ZoneID) REFERENCES ZONE(ZoneID)
);

CREATE TABLE SENSOR_DATA (
  DataID VARCHAR(20) PRIMARY KEY,
  SensorID VARCHAR(20) NOT NULL,
  SensorTypeID VARCHAR(20) NOT NULL,
  MeasuredValue DECIMAL(10,2) NOT NULL,
  MeasuredAt DATETIME NOT NULL,
  CollectionSource VARCHAR(30),
  SyncStatus VARCHAR(20),
  CollectedAt DATETIME,
  CONSTRAINT FK_DATA_SENSOR
    FOREIGN KEY (SensorID) REFERENCES SENSOR_NODE(SensorID),
  CONSTRAINT FK_DATA_TYPE
    FOREIGN KEY (SensorTypeID) REFERENCES SENSOR_TYPE(SensorTypeID)
);

CREATE TABLE THRESHOLD (
  ThresholdID VARCHAR(20) PRIMARY KEY,
  SensorTypeID VARCHAR(20) NOT NULL,
  ZoneID VARCHAR(20) NOT NULL,
  MinValue DECIMAL(10,2),
  MaxValue DECIMAL(10,2),
  TimeoutMinutes INT,
  LowBatteryLevel DECIMAL(5,2),
  IsActive BOOLEAN,
  CONSTRAINT FK_THRESHOLD_TYPE
    FOREIGN KEY (SensorTypeID) REFERENCES SENSOR_TYPE(SensorTypeID),
  CONSTRAINT FK_THRESHOLD_ZONE
    FOREIGN KEY (ZoneID) REFERENCES ZONE(ZoneID)
);

CREATE TABLE UAV (
  UAVID VARCHAR(20) PRIMARY KEY,
  UAVCode VARCHAR(50),
  SerialNumber VARCHAR(50),
  Model VARCHAR(50),
  BatteryLevel DECIMAL(5,2),
  Status VARCHAR(20),
  Latitude DECIMAL(10,8),
  Longitude DECIMAL(11,8),
  CreatedAt DATETIME
);

CREATE TABLE MOBILE_GATEWAY (
  GatewayID VARCHAR(20) PRIMARY KEY,
  UAVID VARCHAR(20) NOT NULL,
  GatewayCode VARCHAR(50),
  SerialNumber VARCHAR(50),
  ConnectivityStatus VARCHAR(20),
  BatteryLevel DECIMAL(5,2),
  StorageStatus VARCHAR(20),
  LastSyncAt DATETIME,
  CONSTRAINT FK_GATEWAY_UAV
    FOREIGN KEY (UAVID) REFERENCES UAV(UAVID)
);

CREATE TABLE MISSION (
  MissionID VARCHAR(20) PRIMARY KEY,
  FarmID VARCHAR(20) NOT NULL,
  UAVID VARCHAR(20),
  GatewayID VARCHAR(20),
  CreatedBy VARCHAR(20),
  MissionName VARCHAR(100),
  Status VARCHAR(20),
  StartTime DATETIME,
  EndTime DATETIME,
  CreatedAt DATETIME,
  CONSTRAINT FK_MISSION_FARM
    FOREIGN KEY (FarmID) REFERENCES FARM(FarmID),
  CONSTRAINT FK_MISSION_UAV
    FOREIGN KEY (UAVID) REFERENCES UAV(UAVID),
  CONSTRAINT FK_MISSION_GATEWAY
    FOREIGN KEY (GatewayID) REFERENCES MOBILE_GATEWAY(GatewayID),
  CONSTRAINT FK_MISSION_USER
    FOREIGN KEY (CreatedBy) REFERENCES USER(UserID)
);

CREATE TABLE MISSION_SENSOR (
  MissionID VARCHAR(20),
  SensorID VARCHAR(20),
  PRIMARY KEY (MissionID, SensorID),
  CONSTRAINT FK_MS_MISSION
    FOREIGN KEY (MissionID) REFERENCES MISSION(MissionID),
  CONSTRAINT FK_MS_SENSOR
    FOREIGN KEY (SensorID) REFERENCES SENSOR_NODE(SensorID)
);

CREATE TABLE WAYPOINT (
  WaypointID VARCHAR(20) PRIMARY KEY,
  MissionID VARCHAR(20) NOT NULL,
  SequenceNo INT NOT NULL,
  Latitude DECIMAL(10,8),
  Longitude DECIMAL(11,8),
  Altitude DECIMAL(8,2),
  Status VARCHAR(20),
  ReachedAt DATETIME,
  CONSTRAINT FK_WAYPOINT_MISSION
    FOREIGN KEY (MissionID) REFERENCES MISSION(MissionID)
);

CREATE TABLE MISSION_COLLECTION (
  CollectionID VARCHAR(20) PRIMARY KEY,
  MissionID VARCHAR(20) NOT NULL,
  SensorID VARCHAR(20) NOT NULL,
  CollectedAt DATETIME,
  Latitude DECIMAL(10,8),
  Longitude DECIMAL(11,8),
  Status VARCHAR(20),
  ErrorReason TEXT,
  RetryCount INT,
  CONSTRAINT FK_COLLECTION_MISSION
    FOREIGN KEY (MissionID) REFERENCES MISSION(MissionID),
  CONSTRAINT FK_COLLECTION_SENSOR
    FOREIGN KEY (SensorID) REFERENCES SENSOR_NODE(SensorID)
);

CREATE TABLE MISSION_NOTE (
  NoteID VARCHAR(20) PRIMARY KEY,
  MissionID VARCHAR(20) NOT NULL,
  UserID VARCHAR(20) NOT NULL,
  NoteContent TEXT,
  CreatedAt DATETIME,
  CONSTRAINT FK_NOTE_MISSION
    FOREIGN KEY (MissionID) REFERENCES MISSION(MissionID),
  CONSTRAINT FK_NOTE_USER
    FOREIGN KEY (UserID) REFERENCES USER(UserID)
);

CREATE TABLE ALERT (
  AlertID VARCHAR(20) PRIMARY KEY,
  SensorID VARCHAR(20) NOT NULL,
  ClosedBy VARCHAR(20),
  AlertType VARCHAR(50),
  Message VARCHAR(255),
  Severity VARCHAR(20),
  TriggeredValue DECIMAL(10,2),
  TriggeredAt DATETIME,
  Status VARCHAR(20),
  ClosedAt DATETIME,
  CONSTRAINT FK_ALERT_SENSOR
    FOREIGN KEY (SensorID) REFERENCES SENSOR_NODE(SensorID),
  CONSTRAINT FK_ALERT_CLOSED_BY
    FOREIGN KEY (ClosedBy) REFERENCES USER(UserID)
);

CREATE TABLE NOTIFICATION (
  NotificationID VARCHAR(20) PRIMARY KEY,
  AlertID VARCHAR(20) NOT NULL,
  UserID VARCHAR(20) NOT NULL,
  NotificationType VARCHAR(20),
  Message VARCHAR(255),
  SentAt DATETIME,
  Status VARCHAR(20),
  CONSTRAINT FK_NOTIFICATION_ALERT
    FOREIGN KEY (AlertID) REFERENCES ALERT(AlertID),
  CONSTRAINT FK_NOTIFICATION_USER
    FOREIGN KEY (UserID) REFERENCES USER(UserID)
);

CREATE TABLE ALERT_HANDLING (
  HandlingID VARCHAR(20) PRIMARY KEY,
  AlertID VARCHAR(20) NOT NULL,
  HandledBy VARCHAR(20) NOT NULL,
  Action VARCHAR(100),
  Note TEXT,
  HandledAt DATETIME,
  CONSTRAINT FK_HANDLING_ALERT
    FOREIGN KEY (AlertID) REFERENCES ALERT(AlertID),
  CONSTRAINT FK_HANDLING_USER
    FOREIGN KEY (HandledBy) REFERENCES USER(UserID)
);

CREATE INDEX IX_SENSOR_DATA_TIME
  ON SENSOR_DATA(SensorID, MeasuredAt);

CREATE INDEX IX_ALERT_STATUS
  ON ALERT(SensorID, Status, TriggeredAt);

CREATE INDEX IX_COLLECTION_STATUS
  ON MISSION_COLLECTION(MissionID, Status);

CREATE INDEX IX_WAYPOINT_ORDER
  ON WAYPOINT(MissionID, SequenceNo);
\end{lstlisting}
