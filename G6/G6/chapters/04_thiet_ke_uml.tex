\section{Activity Diagram: xử lý measurement}

\begin{figure}[H]
\centering
\begin{tikzpicture}[
node distance=0.8cm,
act/.style={draw, rounded corners, align=center, minimum width=4.0cm, minimum height=0.7cm, font=\small},
decision/.style={draw, diamond, aspect=2.1, align=center, font=\small, inner sep=1pt},
arr/.style={-Latex, semithick, font=\scriptsize}
]
\node[act] (start) {Nhận measurement};
\node[act,below=of start] (validate) {Validate format + device};
\node[decision,below=of validate] (valid) {Hợp lệ?};
\node[act,below left=1.0cm and 1.5cm of valid] (reject) {Ghi nhận lỗi / loại record};
\node[act,below right=1.0cm and 1.5cm of valid] (save) {Lưu dữ liệu};
\node[act,below=of save] (threshold) {Kiểm tra threshold};
\node[decision,below=of threshold] (abn) {Bất thường?};
\node[act,below left=1.0cm and 1.5cm of abn] (normal) {Kết thúc xử lý};
\node[act,below right=1.0cm and 1.5cm of abn] (alert) {Tạo cảnh báo};

\draw[arr] (start)--(validate);
\draw[arr] (validate)--(valid);
\draw[arr] (valid)-- node[above left]{Không} (reject);
\draw[arr] (valid)-- node[above right]{Có} (save);
\draw[arr] (save)--(threshold);
\draw[arr] (threshold)--(abn);
\draw[arr] (abn)-- node[above left]{Không} (normal);
\draw[arr] (abn)-- node[above right]{Có} (alert);
\end{tikzpicture}
